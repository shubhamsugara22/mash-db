-- Test script demonstrating SHOW STATS command
-- This script shows the lifecycle of collecting and retrieving statistics

-- Create a table with sample data
CREATE TABLE employees (id INT, name TEXT, department TEXT, salary INT);

-- Insert some data
INSERT INTO employees VALUES (1, 'Alice', 'Engineering', 95000);
INSERT INTO employees VALUES (2, 'Bob', 'Sales', 75000);
INSERT INTO employees VALUES (3, 'Charlie', 'Engineering', 90000);
INSERT INTO employees VALUES (4, 'Diana', 'HR', 70000);
INSERT INTO employees VALUES (5, 'Eve', 'Sales', 78000);

-- Analyze the table to collect statistics
ANALYZE TABLE employees;

-- Show statistics for the employees table
SHOW STATS employees;

-- Show all available statistics in the database
SHOW STATS;
