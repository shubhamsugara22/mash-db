# DISTINCT Feature Demonstration

# Setup: Insert test data with duplicate usernames
insert 1 alice alice@example.com
insert 2 bob bob@example.com
insert 3 alice alice2@example.com
insert 4 charlie charlie@example.com
insert 5 bob bob2@example.com

# Test 1: SELECT * (shows all rows including duplicates)
select *

# Test 2: SELECT DISTINCT * (removes duplicate rows based on all columns)
select distinct *

# Test 3: SELECT DISTINCT username (shows unique usernames)
select distinct username

# Test 4: DISTINCT with WHERE clause
select distinct username where id>2

# Test 5: DISTINCT with ORDER BY
select distinct username order by username asc

# Test 6: DISTINCT with LIMIT
select distinct username limit 2

# Test 7: DISTINCT full clause (WHERE + ORDER BY + LIMIT)
select distinct username where id>1 order by username desc limit 3

# Expected Results:
# Test 1: All 5 rows shown
# Test 2: All 5 rows (no full duplicates exist)
# Test 3: Only unique usernames: alice, bob, charlie
# Test 4: Unique usernames where id>2: alice, charlie, bob
# Test 5: Sorted unique usernames: alice, bob, charlie
# Test 6: First 2 unique usernames: alice, bob (or in insertion order)
# Test 7: Filtered (id>1), sorted DESC, limited: charlie, bob, alice
