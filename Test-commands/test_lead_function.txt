# Test LEAD Window Function

# Create test table
CREATE TABLE sales (id, employee, amount, sale_date)

# Insert test data
INSERT INTO sales VALUES (1, 'Alice', '1000', '2024-01-01')
INSERT INTO sales VALUES (2, 'Bob', '1500', '2024-01-02')
INSERT INTO sales VALUES (3, 'Alice', '2000', '2024-01-03')
INSERT INTO sales VALUES (4, 'Bob', '1200', '2024-01-04')
INSERT INTO sales VALUES (5, 'Alice', '3000', '2024-01-05')

# Test 1: Basic LEAD - show each employee's next sale amount
SELECT employee, amount, LEAD(amount) OVER (ORDER BY id) FROM sales

# Test 2: LEAD with offset 2 - look ahead 2 rows
SELECT employee, amount, LEAD(amount, 2) OVER (ORDER BY id) FROM sales

# Test 3: LEAD with custom default - use 'No More Sales' when no next row
SELECT employee, amount, LEAD(amount, 1, 'No More Sales') OVER (ORDER BY id) FROM sales

# Test 4: LEAD with PARTITION BY - next sale within same employee
SELECT employee, amount, LEAD(amount) OVER (PARTITION BY employee ORDER BY id) FROM sales

# Test 5: LEAD with DESC ordering
SELECT employee, amount, LEAD(amount) OVER (ORDER BY id DESC) FROM sales

# Test 6: Multiple LEAD functions
SELECT employee, amount, LEAD(amount, 1, 'N/A') OVER (ORDER BY id), LEAD(employee) OVER (ORDER BY id) FROM sales
