# Test suite for INNER, LEFT, and RIGHT JOIN functionality

# Test 1: Basic INNER JOIN
# Should return only users that have matching orders
SELECT * FROM users INNER JOIN orders ON id = id

# Test 2: LEFT JOIN - Keep all left table rows
# Should return all users, even those without orders
SELECT * FROM users LEFT JOIN orders ON id = id

# Test 3: RIGHT JOIN - Simplified implementation
# Should return only users that have matching orders
SELECT * FROM users RIGHT JOIN orders ON id = id

# Test 4: INNER JOIN with WHERE clause
# Should filter before joining

# Should project specific columns after LEFT JOIN

# Test 7: JOIN with LIMIT
